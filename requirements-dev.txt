# Development and CI dependencies.
#
# Kept separate from requirements.txt because the Dockerfile installs ONLY
# requirements.txt — so nothing here ships to the Railway production image.
# Before this split, pytest, pytest-asyncio and ruff were all installed in prod.
#
# CI installs both:  pip install -r requirements.txt -r requirements-dev.txt

-r requirements.txt

# 8.4 is the floor pytest-asyncio 1.x requires (`pytest<10,>=8.4`), not a want.
# Staying on 8.x keeps the move to pytest 9 (dependabot #89) a separate,
# deliberate decision — 9.1.1 also passes this suite, it just is not pinned here.
pytest==8.4.2
# 1.x closes the per-test event loop instead of leaving it to the garbage
# collector. Under 0.24 the loop's self-pipe socketpair emitted an "unclosed
# event loop" ResourceWarning at whatever later moment GC ran, and
# filterwarnings=["error"] turned that into a failure in whichever unrelated
# test was executing — it started landing on test_meta_suite once the async
# test count grew. Verified: 0 live sockets after a full run, was 2.
pytest-asyncio==1.4.0
pytest-cov==6.0.0
ruff==0.16.3
# Reads the IRS 990 bulk zips, which use Deflate64 (method 9) — Python's
# zipfile, bsdtar/libarchive and macOS ditto all refuse it, and bsdtar fails
# by writing a zero-byte file rather than erroring. Also streams, so
# scripts/ingest_funding_edges.py abandons a 497MB archive as soon as its
# targets are found. Used only by that script, never by the running service,
# so it belongs here and not in requirements.txt.
stream-unzip==0.0.101
